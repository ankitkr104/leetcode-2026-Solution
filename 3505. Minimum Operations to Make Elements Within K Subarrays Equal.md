# Problem Statement

You are given an integer array `nums` and two integers `x` and `k`.

You can perform the following operation any number of times:

- Increase or decrease any element of `nums` by `1`.

Return the minimum number of operations required to obtain at least `k` non-overlapping subarrays of size exactly `x`, such that all elements within each selected subarray are equal.

## Examples

### Example 1

**Input:**
    nums = [5,-2,1,3,7,3,6,4,-1]
    x = 3
    k = 2

**Output:**
    8

**Explanation:**

We can make these two non-overlapping subarrays equal:

    [1, 1, 1]  // indices 1 to 3
    [4, 4, 4]  // indices 5 to 7

The total operations required are:

    3 + 2 + 1 + 2 = 8

### Example 2

**Input:**
    nums = [9,-2,-2,-2,1,5]
    x = 2
    k = 2

**Output:**
    3

**Explanation:**

We can make the following subarrays equal:

    [-2, -2]  // indices 1 to 2
    [-2, -2]  // indices 3 to 4

Only `nums[4]` needs to be decreased by `3`.

Therefore, the answer is `3`.

## Constraints

- `2 <= nums.length <= 10^5`
- `-10^6 <= nums[i] <= 10^6`
- `2 <= x <= nums.length`
- `1 <= k <= 15`
- `2 <= k * x <= nums.length`

## C++ Solution

    class Solution {
    public:
        long long minOperations(vector<int>& nums, int x, int k) {
            int n = nums.size();
    
            multiset<int> low, high;
            long long sumLow = 0, sumHigh = 0;
    
            auto rebalance = [&]() {
                int target = (x + 1) / 2;
    
                while ((int)low.size() > target) {
                    int val = *low.rbegin();
                    low.erase(prev(low.end()));
                    sumLow -= val;
    
                    high.insert(val);
                    sumHigh += val;
                }
    
                while ((int)low.size() < target && !high.empty()) {
                    int val = *high.begin();
                    high.erase(high.begin());
                    sumHigh -= val;
    
                    low.insert(val);
                    sumLow += val;
                }
            };
    
            auto add = [&](int val) {
                if (low.empty() || val <= *low.rbegin()) {
                    low.insert(val);
                    sumLow += val;
                } else {
                    high.insert(val);
                    sumHigh += val;
                }
                rebalance();
            };
    
            auto remove = [&](int val) {
                auto it = low.find(val);
    
                if (it != low.end()) {
                    low.erase(it);
                    sumLow -= val;
                } else {
                    it = high.find(val);
                    high.erase(it);
                    sumHigh -= val;
                }
    
                rebalance();
            };
    
            vector<long long> cost(n - x + 1);
    
            for (int i = 0; i < n; i++) {
                add(nums[i]);
    
                if (i >= x) {
                    remove(nums[i - x]);
                }
    
                if (i >= x - 1) {
                    long long median = *low.rbegin();
    
                    cost[i - x + 1] =
                        median * low.size() - sumLow +
                        sumHigh - median * high.size();
                }
            }
    
            const long long INF = LLONG_MAX / 4;
    
            vector<long long> prev(n + 1, 0);
            vector<long long> curr(n + 1, INF);
    
            for (int j = 1; j <= k; j++) {
                fill(curr.begin(), curr.end(), INF);
    
                for (int i = 1; i <= n; i++) {
                    curr[i] = curr[i - 1];
    
                    if (i >= x && prev[i - x] != INF) {
                        curr[i] = min(
                            curr[i],
                            prev[i - x] + cost[i - x]
                        );
                    }
                }
    
                prev.swap(curr);
            }
    
            return prev[n];
        }
    };
