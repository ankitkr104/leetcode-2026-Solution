# Problem Statement

You are given a 0-indexed integer array `nums`.

The distinct count of a subarray `nums[i..j]` is the number of unique elements in that subarray.

Return the sum of the squares of the distinct counts of all non-empty contiguous subarrays of `nums`.

Since the answer may be very large, return it modulo `10^9 + 7`.

## Examples

### Example 1

**Input:**
    nums = [1,2,1]

**Output:**
    15

**Explanation:**

The six possible subarrays are:

| Subarray | Distinct Count | Square |
|---|---:|---:|
| [1] | 1 | 1 |
| [2] | 1 | 1 |
| [1] | 1 | 1 |
| [1,2] | 2 | 4 |
| [2,1] | 2 | 4 |
| [1,2,1] | 2 | 4 |

Total:

    1 + 1 + 1 + 4 + 4 + 4 = 15

### Example 2

**Input:**
    nums = [2,2]

**Output:**
    3

**Explanation:**

The three subarrays are `[2]`, `[2]`, and `[2,2]`. Each has one distinct element.

    1² + 1² + 1² = 3

## Constraints

- `1 <= nums.length <= 10^5`
- `1 <= nums[i] <= 10^5`


## C++ Solution

    class Solution {
        static constexpr long long MOD = 1000000007LL;

        struct Node {
            long long sum = 0;
            long long sumSq = 0;
            long long lazy = 0;
        };

        vector<Node> tree;
        int n;

        void apply(int node, int left, int right, long long value) {
            long long length = right - left + 1LL;

            tree[node].sumSq =
                (tree[node].sumSq
                 + 2LL * value * tree[node].sum
                 + length * value % MOD * value) % MOD;

            tree[node].sum =
                (tree[node].sum + length * value) % MOD;

            tree[node].lazy =
                (tree[node].lazy + value) % MOD;
        }

        void push(int node, int left, int right) {
            if (tree[node].lazy == 0 || left == right) {
                return;
            }

            int mid = left + (right - left) / 2;
            long long value = tree[node].lazy;

            apply(node * 2, left, mid, value);
            apply(node * 2 + 1, mid + 1, right, value);

            tree[node].lazy = 0;
        }

        void pull(int node) {
            tree[node].sum =
                (tree[node * 2].sum + tree[node * 2 + 1].sum) % MOD;

            tree[node].sumSq =
                (tree[node * 2].sumSq + tree[node * 2 + 1].sumSq) % MOD;
        }

        void update(int node, int left, int right,
                    int queryLeft, int queryRight) {
            if (queryLeft > right || queryRight < left) {
                return;
            }

            if (queryLeft <= left && right <= queryRight) {
                apply(node, left, right, 1);
                return;
            }

            push(node, left, right);

            int mid = left + (right - left) / 2;

            update(node * 2, left, mid, queryLeft, queryRight);
            update(node * 2 + 1, mid + 1, right,
                   queryLeft, queryRight);

            pull(node);
        }

    public:
        int sumCounts(vector<int>& nums) {
            n = nums.size();
            tree.assign(4 * n, Node());

            vector<int> last(100001, -1);
            long long answer = 0;

            for (int i = 0; i < n; i++) {
                int previous = last[nums[i]];

                update(1, 0, n - 1, previous + 1, i);

                last[nums[i]] = i;

                answer = (answer + tree[1].sumSq) % MOD;
            }

            return static_cast<int>(answer);
        }
    };
