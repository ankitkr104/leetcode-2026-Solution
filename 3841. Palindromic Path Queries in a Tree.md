# Problem Statement

You are given an undirected tree with `n` nodes labeled from `0` to `n - 1`, represented by an edge list `edges`.

You are also given a string `s` of length `n`, where `s[i]` represents the character assigned to node `i`.

You are given a list of queries. Each query is one of two types:

- **Update:** `"update u c"` — change the character at node `u` to `c`.
- **Query:** `"query u v"` — determine whether the characters on the unique path from node `u` to node `v`, inclusive, can be rearranged into a palindrome.

Return a Boolean array containing the results of all queries of type `"query"` in their original order.

A string can be rearranged into a palindrome if **at most one character occurs an odd number of times**.

## Examples

### Example 1

**Input:**
    n = 3
    edges = [[0,1],[1,2]]
    s = "aac"
    queries = ["query 0 2","update 1 b","query 0 2"]

**Output:**
    [true,false]

**Explanation:**

- `"query 0 2"`: The path characters are `"aac"`. Only `a` occurs an odd number of times, so the answer is `true`.
- `"update 1 b"`: The character at node `1` changes to `b`.
- `"query 0 2"`: The path characters are `"abc"`. Three characters occur an odd number of times, so the answer is `false`.

### Example 2

**Input:**
    n = 4
    edges = [[0,1],[0,2],[0,3]]
    s = "abca"
    queries = ["query 1 2","update 0 b","query 2 3","update 3 a","query 1 3"]

**Output:**
    [false,false,true]

## Constraints

- `1 <= n == s.length <= 5 * 10^4`
- `edges.length == n - 1`
- `0 <= ui, vi < n`
- `s` consists of lowercase English letters.
- `edges` represents a valid tree.
- `1 <= queries.length <= 5 * 10^4`
- Queries are either `"update u c"` or `"query u v"`.

## C++ Solution

        class Solution {
        struct SegTree {
            int n;
            vector<int> tree;
    
            SegTree(vector<int>& a) {
                n = (int)a.size();
                tree.assign(4 * n, 0);
                build(1, 0, n - 1, a);
            }
    
            void build(int node, int l, int r, vector<int>& a) {
                if (l == r) {
                    tree[node] = a[l];
                    return;
                }
    
                int mid = (l + r) / 2;
                build(node * 2, l, mid, a);
                build(node * 2 + 1, mid + 1, r, a);
                tree[node] = tree[node * 2] ^ tree[node * 2 + 1];
            }
    
            void update(int node, int l, int r, int idx, int val) {
                if (l == r) {
                    tree[node] = val;
                    return;
                }
    
                int mid = (l + r) / 2;
                if (idx <= mid)
                    update(node * 2, l, mid, idx, val);
                else
                    update(node * 2 + 1, mid + 1, r, idx, val);
    
                tree[node] = tree[node * 2] ^ tree[node * 2 + 1];
            }
    
            int query(int node, int l, int r, int ql, int qr) {
                if (qr < l || r < ql) return 0;
                if (ql <= l && r <= qr) return tree[node];
    
                int mid = (l + r) / 2;
                return query(node * 2, l, mid, ql, qr) ^
                       query(node * 2 + 1, mid + 1, r, ql, qr);
            }
    
            void update(int idx, int val) {
                update(1, 0, n - 1, idx, val);
            }
    
            int query(int l, int r) {
                return query(1, 0, n - 1, l, r);
            }
        };
    
    public:
        vector<bool> palindromePath(int n, vector<vector<int>>& edges,
                                    string s, vector<string>& queries) {
            vector<vector<int>> adj(n);
    
            for (auto& e : edges) {
                adj[e[0]].push_back(e[1]);
                adj[e[1]].push_back(e[0]);
            }
    
            vector<int> parent(n, -1), depth(n, 0), sz(n, 1);
            vector<int> heavy(n, -1), head(n), pos(n);
            vector<int> order;
            order.reserve(n);
    
            // Build parent/depth and traversal order iteratively.
            vector<int> st = {0};
            parent[0] = 0;
    
            while (!st.empty()) {
                int u = st.back();
                st.pop_back();
                order.push_back(u);
    
                for (int v : adj[u]) {
                    if (v == parent[u]) continue;
                    parent[v] = u;
                    depth[v] = depth[u] + 1;
                    st.push_back(v);
                }
            }
    
            // Compute subtree sizes and heavy children.
            for (int i = n - 1; i >= 0; --i) {
                int u = order[i];
                int bestSize = 0;
    
                for (int v : adj[u]) {
                    if (parent[v] != u) continue;
    
                    sz[u] += sz[v];
                    if (sz[v] > bestSize) {
                        bestSize = sz[v];
                        heavy[u] = v;
                    }
                }
            }
    
            // Assign HLD heads and positions.
            int timer = 0;
            vector<pair<int, int>> chains = {{0, 0}};
    
            while (!chains.empty()) {
                auto [start, h] = chains.back();
                chains.pop_back();
    
                for (int u = start; u != -1; u = heavy[u]) {
                    head[u] = h;
                    pos[u] = timer++;
    
                    for (int v : adj[u]) {
                        if (parent[v] == u && v != heavy[u]) {
                            chains.push_back({v, v});
                        }
                    }
                }
            }
    
            vector<int> base(n);
            for (int u = 0; u < n; ++u) {
                base[pos[u]] = 1 << (s[u] - 'a');
            }
    
            SegTree seg(base);
    
            auto pathXor = [&](int u, int v) {
                int mask = 0;
    
                while (head[u] != head[v]) {
                    if (depth[head[u]] < depth[head[v]]) {
                        swap(u, v);
                    }
    
                    mask ^= seg.query(pos[head[u]], pos[u]);
                    u = parent[head[u]];
                }
    
                if (depth[u] > depth[v]) swap(u, v);
    
                mask ^= seg.query(pos[u], pos[v]);
                return mask;
            };
    
            vector<bool> ans;
    
            for (const string& q : queries) {
                string type, third;
                int u;
    
                stringstream ss(q);
                ss >> type >> u >> third;
    
                if (type == "update") {
                    char c = third[0];
                    s[u] = c;
                    seg.update(pos[u], 1 << (c - 'a'));
                } else {
                    int v = stoi(third);
                    int mask = pathXor(u, v);
                    ans.push_back((mask & (mask - 1)) == 0);
                }
            }
    
            return ans;
        }
    };
